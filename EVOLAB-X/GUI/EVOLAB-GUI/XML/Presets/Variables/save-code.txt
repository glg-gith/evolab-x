int MainWindow::printCWD() {

    char cCurrentPath[FILENAME_MAX];

    if (!GetCurrentDir(cCurrentPath, sizeof(cCurrentPath)))
    {
        return errno;
    }

    cCurrentPath[sizeof(cCurrentPath) - 1] = '\0'; /* not really required */

    cout << "The current working directory is: " << cCurrentPath << endl;

    return 1;
}

std::string MainWindow::ReplaceAll(std::string str, const std::string& from, const std::string& to) {
    //if(from.empty()) return 0;
    size_t start_pos = 0;
    while((start_pos = str.find(from, start_pos)) != std::string::npos) {
        str.replace(start_pos, from.length(), to);
        start_pos += to.length(); // In case 'to' contains 'from', like replacing 'x' with 'yx'
    }
    return str;
}

void MainWindow::StartOutputCapture()
{
    this->output_cursor = ui->textEdit_2->textCursor();

    //remove((string(TEMP_PROJECT_PATH) + "log.txt").c_str());

    std::streambuf * cout_old = std::cout.rdbuf(cout_buffer.rdbuf());
    std::streambuf * cerr_old = std::cerr.rdbuf(cerr_buffer.rdbuf());
    std::streambuf * clog_old = std::clog.rdbuf(clog_buffer.rdbuf());
}
void MainWindow::EndOutputCapture()
{
    cout.seekp(0);
    cerr.seekp(0);
    clog.seekp(0);
    cout_buffer.seekp(0);
    cerr_buffer.seekp(0);
    clog_buffer.seekp(0);
    cout_buffer.str("");
    cerr_buffer.str("");
    clog_buffer.str("");
    cout.clear();
    cerr.clear();
    clog.clear();
    cout_buffer.clear();
    cerr_buffer.clear();
    clog_buffer.clear();
}
QString MainWindow::GetCapturedOutput()
{
    QString qtext = "";

    std::string str1, str2, str3;
    //    while (std::getline(cout_buffer, str)) { qtext += infoHtml + QString::fromStdString(str) + endHtml; }
    //    while (std::getline(cerr_buffer, str)) { qtext += errorHtml + QString::fromStdString(str) + endHtml; }
    //    while (std::getline(clog_buffer, str)) { qtext += criticalHtml + QString::fromStdString(str) + endHtml; }

    while (std::getline(cout_buffer, str1) || std::getline(cerr_buffer, str2) || std::getline(clog_buffer, str3)) {
        if (str1.size() != 0) qtext += infoHtml + QString::fromStdString(str1) + endHtmlCR;
        if (str2.size() != 0) qtext += errorHtml + QString::fromStdString(str2) + endHtmlCR;
        if (str3.size() != 0) qtext += criticalHtml + QString::fromStdString(str3) + endHtmlCR;
    }

    // Reset for next captures
    EndOutputCapture();
    StartOutputCapture();

    return qtext;
}

void MainWindow::outputTimeoutSlot()
{
    QString qstr = GetCapturedOutput();
    if (!qstr.isEmpty())          // Avoid to monopolize cursor (moved to end) if nothing to be printed
    {
        this->output_cursor.movePosition(QTextCursor::End);
        // Force scrolling to end of text
        ui->textEdit_2->setTextCursor(this->output_cursor);
        ui->textEdit_2->insertHtml(qstr);
        this->output_cursor.movePosition(QTextCursor::End);
        // Force scrolling to end of text
        ui->textEdit_2->setTextCursor(this->output_cursor);

        // Log to file as well...
        std::ofstream myfile;
        myfile.open ( (string(TEMP_PROJECT_PATH) + "log.txt").c_str(), ios::out | ios::app );
        QString s = qstr.replace("<br/>", "\n");
        myfile << s.remove(QRegExp("<[^>]*>")).toStdString() << endl;
        myfile.close();
    }
}

{

    for (int i = 0; i < ui->listWidget->selectedItems().size(); i++)

        l.push_back(ui->listWidget->row(ui->listWidget->selectedItems().at(i)));
    l.sort();
    std::list<int>::reverse_iterator it;
    for (it = l.rbegin(); it != l.rend(); ++it) {
        ui->listWidget->takeItem(*it);
    }
}

void MainWindow::clearLayout(QLayout* layout, bool deleteWidgets = true)
{
    while (QLayoutItem* item = layout->takeAt(0))
    {
        if (deleteWidgets)
        {
            if (QWidget* widget = item->widget())
                delete widget;
        }
        if (QLayout* childLayout = item->layout())
            clearLayout(childLayout, deleteWidgets);
        delete item;
    }
}

void MainWindow::UpdatePopCell(int row, int col, PointerLessModels::t_cell_items_list *model)
{
    stringstream ss;
    ss << "Updated pop cell at [" << row << "," << col << "]" << endl;
    qDebug(ss.str().c_str());

    QTableWidgetItem *item = ui->tableWidget_Pop->item(row, col);

    if (ui->tableWidget_Pop->cellWidget(row, col) == NULL) {
        qDebug("Create Combobox!");
    } else {
        qDebug("Update Combobox!");
        qDebug(std::to_string(model->items.size()).c_str());

        // Rebuild combobox
        int default_row = -1;
        QComboBox *cb = (QComboBox *)ui->tableWidget_Pop->cellWidget(row, col);
        cb->clear();
        for (unsigned int i=0; i < model->items.size(); i++) {
            PointerLessModels::t_basic_item an_item = model->items.at(i);
            cb->addItem(QString::fromStdString(an_item.name));
            // Try finding the default if applicable
            for (unsigned int j=0; j < an_item.params.size(); ++j) {
                if (an_item.params.at(j).name == "Is Default" &&
                        an_item.params.at(j).value == "1") { default_row = i; }
            }
        }

        // Set default icon
        if (default_row != -1) { cb->setItemIcon(default_row, this->CHK_ICON); cb->setCurrentIndex(default_row); }

        qDebug("Combobox updated!");
    }
}

{
    QTableWidgetItem *item = ui->tableWidget_Pop->currentItem();
    if (item->column() != 0)
    {

        this->popCellClipboard = *(PointerLessModels::t_cell_items_list *)(item->data(Qt::UserRole).value<quintptr>());
    }
}
// Cut cell
void MainWindow::on_toolButton_CutPopCellItem_clicked()
{
    QTableWidgetItem *item = ui->tableWidget_Pop->currentItem();
    if (item->column() != 0)
    {
        // Copy current to clipboard
        PointerLessModels::t_cell_items_list *p_cil = (PointerLessModels::t_cell_items_list *)(item->data(Qt::UserRole).value<quintptr>());
        this->popCellClipboard = *p_cil;

        // Clear current (if not locked = modif protected)
        if (!this->IsLockedPopCellItem(item))
        {
            *p_cil = PointerLessModels::t_cell_items_list();
            this->UpdatePopCell(item->row(), item->column(), p_cil);
            this->UpdatePopCellSettings(item->row(), item->column(), p_cil);
        }
    }
}
// Paste cell
void MainWindow::on_toolButton_PastePopCellItem_clicked()
{
    QTableWidgetItem *item = ui->tableWidget_Pop->currentItem();

    // Paste (if not locked = modif protected)
    if (item->column() != 0 && !this->IsLockedPopCellItem(item))
    {
        // Skip non-compatible copy (Cell must be of same type!)
        if (this->popCellClipboard.type != (PointerLessModels::CellType)(item->column()-1)) return;

        PointerLessModels::t_cell_items_list *pasted_cal = new PointerLessModels::t_cell_items_list();
        *pasted_cal = this->popCellClipboard;

        item->setData(Qt::UserRole, QVariant::fromValue(reinterpret_cast<quintptr>(pasted_cal)));

        this->UpdatePopCell(item->row(), item->column(), pasted_cal);
        this->refreshWidget(ui->tableWidget_Pop);
    }
}

void MainWindow::MoveCurrentPopCellItem(int step)
{
    cerr << "on_toolButton_PopCellItemUp_clicked()" << endl;

    /** Swap up **/
    int cur_idx = ui->listWidget_ItemList->currentRow();

    // Remove item
    QListWidgetItem *item = ui->listWidget_ItemList->takeItem(cur_idx);
    PointerLessModels::t_basic_item *p_item = (PointerLessModels::t_basic_item *)(item->data(Qt::UserRole).value<quintptr>());
    QTableWidgetItem *cell = ui->tableWidget_Pop->currentItem();
    PointerLessModels::t_cell_items_list *p_cil = (PointerLessModels::t_cell_items_list *)(cell->data(Qt::UserRole).value<quintptr>());
    std::vector<PointerLessModels::t_basic_item>::iterator it = p_cil->items.begin() + cur_idx; //std::advance(it, cur_idx); // (Old compiler version)
    p_cil->items.erase(it);

    // Re-insert item
    cur_idx += step; it += step; // Step back (step < 0) or forward (step > 0)
    p_cil->items.insert(it, *p_item);
    //ui->listWidget_ItemList->setCurrentRow(cur_idx);
    ui->listWidget_ItemList->insertItem(cur_idx, item);
    ui->listWidget_ItemList->setCurrentRow(cur_idx);

    // Update tableWidget
    ui->listWidget_ItemList->EmitUpdateForeignModelSignal(cur_idx, p_item);
}

}

{
    cerr << "on_toolButton_PopCellItemUp_clicked()" << endl;

std::vector<PointerLessModels::t_pop> *MainWindow::GetPopsList()
{
    std::vector<PointerLessModels::t_cells_group> *pops_vec = new std::vector<PointerLessModels::t_cells_group>();
    for (unsigned int i=0; i < ui->tableWidget_Pop->rowCount(); ++i)
    {
        PointerLessModels::t_cells_group a_pop;// = new PointerLessModels::t_pop();
        QTableWidgetItem *item0 = ui->tableWidget_Pop->item(i, 0);                  // Col 0
        PointerLessModels::t_basic_text *p_text = (PointerLessModels::t_basic_text *)item0->data(Qt::UserRole).value<quintptr>();
        a_pop.name = p_text->name;
        a_pop.desc = p_text->desc;
        QTableWidgetItem *item;
        for (unsigned int j=1; j < ui->tableWidget_Pop->columnCount(); ++j) {       // Other cols
            item = ui->tableWidget_Pop->item(i, j);
            PointerLessModels::t_cell_items_list *p_cil = (PointerLessModels::t_cell_items_list *)item->data(Qt::UserRole).value<quintptr>();
            a_pop.cells.push_back(*p_cil);
        }
        pops_vec->push_back(a_pop);
    }
    return pops_vec;
}

void MainWindow::MoveCurrentPop(int step)
{
    cerr << "on_toolButton_PopRowUp_clicked()" << endl;

    // Disconnect
    this->disconnect(ui->tableWidget_Pop, SIGNAL(currentItemChanged(QTableWidgetItem*,QTableWidgetItem*)), this, SLOT(on_tableWidget_Pop_currentItemChanged(QTableWidgetItem*,QTableWidgetItem*)));

    /** Swap up/down **/
    int row = ui->tableWidget_Pop->currentRow();
    int col = ui->tableWidget_Pop->currentColumn();
    int cur_idx = row;
    bool move_up = (step < 0);
    int next_idx = cur_idx + step;

    // Save both rows data
    PointerLessModels::t_cells_group cur_cg, next_cg;
    // Save "is_locked"
    std::vector<bool> cur_islocked, next_islocked;

    // Cur
    PointerLessModels::t_basic_text *p_cur_text = (PointerLessModels::t_basic_text *)ui->tableWidget_Pop->item(cur_idx, 0)->data(Qt::UserRole).value<quintptr>();
    cur_cg.name = p_cur_text->name;
    cur_cg.desc = p_cur_text->desc;
    for (unsigned int i=1; i < ui->tableWidget_Pop->columnCount(); i++)
    {
        PointerLessModels::t_cell_items_list *p_cil = (PointerLessModels::t_cell_items_list *)ui->tableWidget_Pop->item(cur_idx, i)->data(Qt::UserRole).value<quintptr>();
        cur_cg.cells.push_back(*p_cil);

        bool is_locked = this->IsLockedPopCellItem(ui->tableWidget_Pop->item(cur_idx, i));
        cur_islocked.push_back(is_locked);
        //        if (is_locked) {
        //            std::vector<QTableWidgetItem *>::iterator it = std::find(this->lockedPopCells.begin(), this->lockedPopCells.end(), ui->tableWidget_Pop->item(cur_idx, i));
        //            this->lockedPopCells.erase(it);
        //        }
        this->SetLockedPopCellItem(ui->tableWidget_Pop->item(cur_idx, i), is_locked);
    }
    // Next
    PointerLessModels::t_basic_text *p_next_text = (PointerLessModels::t_basic_text *)ui->tableWidget_Pop->item(next_idx, 0)->data(Qt::UserRole).value<quintptr>();
    next_cg.name = p_next_text->name;
    next_cg.desc = p_next_text->desc;
    for (unsigned int i=1; i < ui->tableWidget_Pop->columnCount(); i++)
    {
        PointerLessModels::t_cell_items_list *p_cil = (PointerLessModels::t_cell_items_list *)ui->tableWidget_Pop->item(next_idx, i)->data(Qt::UserRole).value<quintptr>();
        next_cg.cells.push_back(*p_cil);

        bool is_locked = this->IsLockedPopCellItem(ui->tableWidget_Pop->item(next_idx, i));
        next_islocked.push_back(is_locked);
        if (is_locked) {
            std::vector<QTableWidgetItem *>::iterator it = std::find(this->lockedPopCells.begin(), this->lockedPopCells.end(), ui->tableWidget_Pop->item(next_idx, i));
            this->lockedPopCells.erase(it);
        }
    }

    // Remove rows
    int remove_idx = (move_up) ? next_idx : cur_idx;
    ui->tableWidget_Pop->removeRow(remove_idx);
    ui->tableWidget_Pop->removeRow(remove_idx);

    // Insert rows at
    this->AddPop(&cur_cg, ((move_up) ? next_idx : cur_idx));
    this->AddPop(&next_cg, cur_idx);

    // Restore "is_locked"
    for (unsigned int i=0; i < cur_islocked.size(); i++)
    {
        // Cur
        //        if (next_islocked.at(i)) {
        //            QTableWidgetItem *cur_item = ui->tableWidget_Pop->item(cur_idx, i+1);
        //            cur_item->setIcon(LOCK_ICON);
        //            this->lockedPopCells.push_back(cur_item);
        //        }
        this->SetLockedPopCellItem(ui->tableWidget_Pop->item(cur_idx, i+1), next_islocked.at(i));
        // Next
        //        if (cur_islocked.at(i)) {
        //            QTableWidgetItem *next_item = ui->tableWidget_Pop->item(next_idx, i+1);
        //            next_item->setIcon(LOCK_ICON);
        //            this->lockedPopCells.push_back(next_item);
        //        }
        this->SetLockedPopCellItem(ui->tableWidget_Pop->item(next_idx, i+1), cur_islocked.at(i));
    }

    // Reconnect
    this->connect(ui->tableWidget_Pop, SIGNAL(currentItemChanged(QTableWidgetItem*,QTableWidgetItem*)), this, SLOT(on_tableWidget_Pop_currentItemChanged(QTableWidgetItem*,QTableWidgetItem*)));
    ui->tableWidget_Pop->setCurrentCell(next_idx, col);
    ui->tableWidget_Pop->selectRow(next_idx);
}

void MainWindow::SetLockedPopCellItem(QTableWidgetItem *item, bool locked)
{
    if (this->IsLockedPopCellItem(item) && !locked) {
        std::vector<QTableWidgetItem *>::iterator it = std::find(this->lockedPopCells.begin(), this->lockedPopCells.end(), item);
        this->lockedPopCells.erase(it);
        item->setIcon(EDIT_ICON);
    } else if (!this->IsLockedPopCellItem(item) && locked) {
        this->lockedPopCells.push_back(item);
        item->setIcon(LOCK_ICON);
    }
}

void MainWindow::UpdateNetPopRepartitionTotal()
{
    QString qtext = this->beginDoc;
    QString qtotal;

    // Update total (and check if OK -> switch from Red - "DarkRed / Crimson", to Green - "MediumSeaGreen")
    double total = 0.0f;
    for (int i=0; i < ui->tableWidget_NetPopsRepartition->rowCount(); ++i) {
        total += ((QDoubleSpinBox *)ui->tableWidget_NetPopsRepartition->cellWidget(i, 1))->value();
    }


    if ((ui->comboBox_NetPopsSpread->currentIndex() == 0 && (total > 1.0f-DOUBLE_MARGIN && total < 1.0f+DOUBLE_MARGIN))                                          // Total OK
            || (ui->comboBox_NetPopsSpread->currentIndex() == 1 && total == this->p_net_params->nbnodes))
    {
        qtext += this->greenHtml;
        ui->label_NetPopIcon->setPixmap(OK_ICON.pixmap(24, 24));
        ui->label_NetPopIcon->setToolTip(qtext + "Total value is OK !" + this->endHtml + this->endDoc);
        ui->textEdit_NetPopTotalInfo->clear();
        ui->frame_NetPopTotalInfo->hide();
        this->totalOK = true;
    } else {                                                                                                        // Total KO
        qtext += this->redHtml;
        ui->label_NetPopIcon->setPixmap(KO_ICON.pixmap(24, 24));
        QString tooltip = qtext + QString("Total value should be equal to \"")
                + QString(((ui->comboBox_NetPopsSpread->currentIndex() == 0) ? "1.0\"" : string(std::to_string(this->p_net_params->nbnodes) + "\" (Number of nodes)").c_str()))
                + QString(" !") + this->endHtml + this->endDoc;
        ui->label_NetPopIcon->setToolTip(tooltip);
        ui->textEdit_NetPopTotalInfo->setText(ui->label_NetPopIcon->toolTip());
        ui->frame_NetPopTotalInfo->show();
        this->totalOK = false;
    }

    if (ui->comboBox_NetPopsSpread->currentIndex() == 0)
        qtotal = QString(std::to_string(total).c_str());
    else
        qtotal = QString(std::to_string(RND(total)).c_str());

    qtext += qtotal + this->endHtmlCR + this->endDoc;

    ui->textEdit_SpreadTotal->setText(qtext);

    ui->pushButton_NetGenerate->setEnabled(this->totalOK);
}

void MainWindow::UpdateNetPopRepartition(int index)
{
    //ui->tableWidget_NetPopsRepartition->clear(); Doesn't clear the widgets (widget take ownership of the item, not the invert...)
    ui->tableWidget_NetPopsRepartition->setRowCount(0); // Supposed to discard everything (to be verified...)
    for (unsigned int i=0; i < this->p_meta_model->pops.size(); ++i)
    {
        ui->tableWidget_NetPopsRepartition->insertRow(i);

        // Pop label
        QTableWidgetItem *lbl_item = new QTableWidgetItem();
        ///lbl_item->setText(ui->tableWidget_Pop->item(i, 0)->text());
        lbl_item->setText(this->p_meta_model->pops.at(i).name.c_str());
        lbl_item->setFlags(lbl_item->flags() & Qt::ItemIsEditable);
        ui->tableWidget_NetPopsRepartition->setItem(i, 0, lbl_item);

        // Distribution widget
        QTableWidgetItem *val_item = new QTableWidgetItem();
        val_item->setFlags(val_item->flags() & Qt::ItemIsEditable);     // Disable editing
        ui->tableWidget_NetPopsRepartition->setItem(i, 1, val_item);
        QDoubleSpinBox *spin_box = new QDoubleSpinBox(ui->tableWidget_NetPopsRepartition);
        switch (index) {
        case 0:             // Stochastic spreading
            //  Range [-1.0, 1.0]
            spin_box->setValue(this->p_meta_model->spreadvalues.at(i));
            spin_box->setMinimum(0.0f);
            spin_box->setMaximum(1.0f);
            spin_box->setSingleStep(0.1f);
            spin_box->setDecimals(NB_DECIMALS);//spin_box
            break;
        case 1:             // Fixed spreading
            // Turn QDoubleSpinBox into kinda QSpinBox... Range [0, Infinity]
            ///spin_box->setValue(0.0f);
            spin_box->setValue(this->p_meta_model->spreadvalues.at(i));
            spin_box->setMinimum(0.0f);
            spin_box->setMaximum(std::numeric_limits<double>::max());
            spin_box->setSingleStep(1.0f);
            spin_box->setDecimals(0);
            break;
        default:
            break;
        }
        ui->tableWidget_NetPopsRepartition->setCellWidget(i, 1, spin_box);
        // Connection
        this->connect(spin_box, SIGNAL(valueChanged(double)), this, SLOT(spinbox_NetPopsRepartition_valueChanged(double)));
    }
    this->UpdateNetPopRepartitionTotal();
}


void MainWindow::on_comboBox_NetPopsSpread_currentIndexChanged(int index)
{
    ui->textEdit_SpreadTotal->clear();
    this->UpdateNetPopRepartition(index);
}
