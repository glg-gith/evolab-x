Detailed Description

Basic LabSet iterator utility.

USAGE :

LabSet ls;
int val;
LabSet::iter it(ls);
while ((val = it.next()) != it.end())
{
cout << (val) << " "; // Do something...
}

Or:

LabSet ls;
LabSet::iter it(ls);
for (int val=it.next(); val != it.end(); val=it.next())
{
cout << (val) << " "; // Do something...
}
