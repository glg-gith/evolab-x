532  static void windows_print_stacktrace(CONTEXT* context)
533  {
534  SymInitialize(GetCurrentProcess(), 0, true);
535 
536  STACKFRAME frame = { 0 };
537 
538  /* setup initial stack frame */
539  frame.AddrPC.Offset = context->Eip;
540  frame.AddrPC.Mode = AddrModeFlat;
541  frame.AddrStack.Offset = context->Esp;
542  frame.AddrStack.Mode = AddrModeFlat;
543  frame.AddrFrame.Offset = context->Ebp;
544  frame.AddrFrame.Mode = AddrModeFlat;
545 
546  while (StackWalk(IMAGE_FILE_MACHINE_I386 ,
547  GetCurrentProcess(),
548  GetCurrentThread(),
549  &frame,
550  context,
551  0,
552  SymFunctionTableAccess,
553  SymGetModuleBase,
554  0 ) )
555  {
556  addr2line(__argv[0], (void*)frame.AddrPC.Offset);
557  }
558
